### 동결 점검 한 줄 (세중님 D-1) — **줄 넣기만으로는 안 됩니다**

```
job_posts 는 RLS 가 켜져 있고(ALTER TABLE … ENABLE ROW LEVEL SECURITY) 정책이 셋입니다 —
  「공고는 누구나 읽습니다」 SELECT  ((not hidden and not hold) or is_admin())
  「공고 수정은 관리자만」   UPDATE  is_admin()
  「공고 삭제는 관리자만」   DELETE  is_admin()
**INSERT 정책이 없습니다.** 그리고 `authenticated` 에 **테이블 INSERT 권한도 없습니다**
(칸별 SELECT·UPDATE 와 테이블 DELETE 만 있습니다 · sql/기준구조_1030.sql 24261~).
→ 관리자 화면에서 `insert` 를 보내면 **막힙니다.** 길은 둘입니다 —
   ㉠ INSERT 정책 + 칸별 INSERT 권한을 **새로 주기**  (권한을 넓히는 쪽 · 추천 안 함)
   ㉡ **창구 함수 하나**(security definer)로 넣기      ← 이쪽을 올립니다
★ 선례가 이미 있습니다 — `담당자공고쓰기()`(출처 BIZ · 지금 4건)가 **같은 꼴**입니다.
  그 함수를 베끼면 새로 만드는 것은 **함수 하나**뿐입니다.
```

### ㉵-2 세중님 조건 — 「담당자 공고(BIZ)와 구분할 칸이 **기존 구조에 있나**」 → **있습니다**

```

㉠ 등록자 id   **있습니다 · 이걸로 갑니다** — `evidence` jsonb 가 이미 담고 있습니다.

㉢ 뽑은값      `뽑은값`·`뽑은법` 은 **수집기가 뽑은 값**을 적는 자리입니다

  → 아래 전문에서 그 두 칸을 **표시로 쓰지 않습니다** (넣기는 하지만 덮일 값으로 봅니다).
```

### 전문 (아직 안 올렸습니다)

```sql
-- 관리자가 손으로 공고를 올리는 창구. 담당자공고쓰기() 를 베꼈습니다(2026-10-30 세중님 D).
-- 다른 점 — ① 관리자만(is_admin) ② 기관을 기관번호로 받습니다(ykiho) ③ detail.등록갈래='관리자'
create or replace function public."관리자공고쓰기"(
  "p_기관번호" text, "p_제목" text, "p_직군" text,
  "p_고용형태" text default null, "p_근무지" text default null, "p_인원" int default null,
  "p_접수부터" date default null, "p_접수까지" date default null,
  "p_마감시각" time default null, "p_주소" text default null,
  "p_상세" jsonb default '{}'::jsonb
) returns text
language plpgsql security definer set search_path to 'public'
as $$
declare v uuid := auth.uid(); o record; v_id text;
begin
  if v is null then raise exception '로그인이 필요합니다' using errcode = '42501'; end if;
  if not is_admin() then raise exception '관리자만 올릴 수 있습니다' using errcode = '42501'; end if;

  if coalesce(btrim(p_제목), '') = '' then
    raise exception '제목을 적어 주세요' using errcode = '22023'; end if;
  if p_직군 is null or p_직군 not in ('작업치료사', '물리치료사') then
    raise exception '직군은 작업치료사 또는 물리치료사입니다' using errcode = '22023'; end if;
  if coalesce(btrim(p_주소), '') = '' then
    raise exception '원문 주소를 적어 주세요' using errcode = '22023'; end if;   -- D-3 「원문 주소」
  if p_접수까지 is not null and p_접수부터 is not null and p_접수까지 < p_접수부터 then
    raise exception '마감일이 접수 시작보다 빠릅니다' using errcode = '22023'; end if;

  select * into o from "기관" where "기관번호" = p_기관번호;
  if o is null then raise exception '없는 기관입니다 (기관 표에 줄이 없습니다)' using errcode = '22023'; end if;

  v_id := 'BIZ' || lpad(nextval('"담당자공고번호"')::text, 6, '0');

  insert into job_posts (id, source, external_id, org_name, title,
                         employ_type, work_place, sido, sgg, headcount,
                         apply_from, apply_to, apply_to_time, posted_at, url,
                         job_group, tab, "기관번호", "연결상태", "연결근거",
                         detail, evidence)
  values (v_id, 'BIZ', v_id, o."이름", btrim(p_제목),
          nullif(btrim(coalesce(p_고용형태, '')), ''),
          nullif(btrim(coalesce(p_근무지, '')), ''),
          o."시도", o."시군구", p_인원,
          coalesce(p_접수부터, (now() at time zone 'Asia/Seoul')::date),
          p_접수까지, p_마감시각,
          (now() at time zone 'Asia/Seoul')::date, btrim(p_주소),
          p_직군, o."탭", p_기관번호,
          /* ★ 이 두 칸은 기관잇기() 가 다시 계산합니다 — **표시로 쓰지 않습니다** (㉵-2) */
          '담당자', '관리자가 직접 올린 공고',
          coalesce(p_상세, '{}'::jsonb) || jsonb_build_object('등록갈래', '관리자'),
          /* ★ 구분 칸 — 담당자 공고에는 '공식확인', 관리자 공고에는 '등록갈래' (㉵-2) */
          jsonb_build_object('올린사람', v, '올린때', now(), '등록갈래', '관리자'))
  ;   /* 시험자료·hidden·hold 는 기본값(false) 그대로 — 담당자공고쓰기() 와 같은 꼴 */
  return v_id;
end $$;

revoke all on function public."관리자공고쓰기"(text,text,text,text,text,int,date,date,time,text,jsonb) from public, anon;
grant execute on function public."관리자공고쓰기"(text,text,text,text,text,int,date,date,time,text,jsonb) to authenticated, service_role;
```
