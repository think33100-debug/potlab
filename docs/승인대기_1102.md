## ㉠ 문의 표와 창구 셋 — 앱 안 「문의하기」 폼 (세중님 11/02 · 11/10 전)

```
왜      지금 「문의하기」는 mailto 입니다 — 누르면 메일이 Google 워크스페이스로 갑니다.
        그래서 처리방침의 국외 이전 「거부 방법」을 「앱 안 문의하기 이용」이라고 적을 수 없습니다.
        폼으로 받아 Supabase(서울)에 두면 그 이전이 없어집니다.
화면     web/app/contact/page.tsx 에 이미 만들어 뒀습니다 (가지 ui/1102-오픈손질).
        `web/lib/inquiry.ts` 의 `문의폼_켰나 = false` 라 **회원에게는 아직 안 보입니다** —
        이 SQL 이 올라간 뒤 그 한 줄을 true 로 바꾸면 켜집니다
        (미리보기에서는 주소 끝에 `?폼=1` 을 붙이면 보입니다)
관리자    web/app/admin/inquiries/page.tsx — 목록 · 상태(받음·처리중·끝) · 메모
위험     **낮음**. 새 표 하나 · 새 함수 셋 · 기존 표·함수·크론 건드리지 않음.
        회원 화면에 새로 보이는 것은 **폼 한 자리**이고 그것도 스위치가 꺼져 있습니다
되돌림    drop function 셋 → drop table 문의  (아래 꼬리에 전문)
        ★ `drop` 글자가 있으니 되돌릴 때 Supabase 승인 창이 뜹니다 (지침 8-7)
```

### 전문 ㉠ (그대로 올립니다)

```sql
-- 2026-11-02 · 앱 안 문의 폼 — 표 하나 + 창구 셋
-- 받는 사람: 회원(authenticated)만. 비회원은 지금처럼 메일로 받습니다.
--   (비회원에게 열면 아이피를 못 세어 스팸을 막을 길이 없습니다)
begin;

create table if not exists "문의" (
  id          bigserial primary key,
  profile_id  uuid        not null references profiles(id) on delete cascade,
  "갈래"      text        not null
                check ("갈래" in ('일반', '개인정보', '가림요청', '신고이의')),
  "글"        text        not null check (length(btrim("글")) between 5 and 4000),
  "공고주소"  text        check ("공고주소" is null or length("공고주소") <= 500),
  "답받을메일" text       check ("답받을메일" is null or "답받을메일" ~ '^[^@[:space:]]+@[^@[:space:]]+$'),
  "상태"      text        not null default '받음'
                check ("상태" in ('받음', '처리중', '끝')),
  "처리메모"  text,
  "처리한사람" uuid,
  "처리한때"  timestamptz,
  "낸때"      timestamptz not null default now(),
  -- 가림요청이면 공고 주소를 반드시 받습니다 (그게 없으면 처리를 못 합니다)
  constraint "가림요청은주소필요"
    check ("갈래" <> '가림요청' or coalesce(btrim("공고주소"), '') <> '')
);

comment on table "문의" is
  '앱 안 「문의하기」로 들어온 글. 회원만 넣습니다(창구 문의넣기). '
  '회원 화면은 이 표를 직접 안 읽습니다 — 관리자 창구 둘로만 봅니다. '
  '보관 기간은 처리방침과 같습니다 — 답을 드린 뒤 1년(치우는 크론은 승인 자리 ㉢).';

create index if not exists "문의_낸때"  on "문의" ("낸때" desc);
create index if not exists "문의_상태"  on "문의" ("상태", "낸때" desc);
create index if not exists "문의_회원"  on "문의" (profile_id, "낸때" desc);

alter table "문의" enable row level security;
-- 정책을 **하나도 만들지 않습니다**. 표에 직접 닿는 길을 아예 두지 않고
-- security definer 창구로만 드나듭니다 (우리 규칙 · 지침 6-2c 와 같은 결)
revoke all on table "문의" from public, anon, authenticated;
revoke all on sequence "문의_id_seq" from public, anon, authenticated;

-- ── 넣기 ──────────────────────────────────────────────────────
-- 하루 상한 5건. 같은 글을 두 번 눌러도 1분 안이면 한 건으로 봅니다
create or replace function "문의넣기"(
  "p_갈래" text, "p_글" text, "p_공고주소" text default null, "p_답받을메일" text default null)
returns bigint
language plpgsql volatile security definer set search_path to 'public'
as $$
declare me uuid := auth.uid(); v_id bigint; v_오늘 int;
begin
  if me is null then
    raise exception '로그인한 분만 보낼 수 있습니다' using errcode = '42501';
  end if;
  select count(*) into v_오늘 from "문의"
   where profile_id = me and "낸때" > now() - interval '1 day';
  if v_오늘 >= 5 then
    raise exception '하루 다섯 건까지 보낼 수 있습니다' using errcode = '53400';
  end if;
  if exists (select 1 from "문의"
              where profile_id = me and btrim("글") = btrim("p_글")
                and "낸때" > now() - interval '1 minute') then
    -- 두 번 눌린 것으로 보고 조용히 앞 건을 돌려줍니다
    select id into v_id from "문의"
     where profile_id = me and btrim("글") = btrim("p_글")
     order by "낸때" desc limit 1;
    return v_id;
  end if;
  insert into "문의" (profile_id, "갈래", "글", "공고주소", "답받을메일")
  values (me, "p_갈래", btrim("p_글"), nullif(btrim("p_공고주소"), ''), nullif(btrim("p_답받을메일"), ''))
  returning id into v_id;
  return v_id;
end $$;

-- ── 관리자 목록 ───────────────────────────────────────────────
create or replace function "admin_문의목록"("p_처리전만" boolean default true, "p_page" int default 0)
returns table(id bigint, "갈래" text, "글" text, "공고주소" text, "답받을메일" text,
              "상태" text, "처리메모" text, "처리한때" timestamptz,
              "낸때" timestamptz, "낸사람" text)
language plpgsql stable security definer set search_path to 'public'
as $$
declare n int;
begin
  if not "권한있나"('신고') then
    raise exception '권한이 없습니다' using errcode = '42501';
  end if;
  select count(*) into n from "문의" q
   where (not "p_처리전만" or q."상태" <> '끝');
  -- 회원이 쓴 글을 모아 보는 화면이라 접속기록을 남깁니다 (법정 기록)
  perform "접속기록남기기"('조회', '문의', null, n);
  return query
    select q.id, q."갈래", q."글", q."공고주소", q."답받을메일",
           q."상태", q."처리메모", q."처리한때", q."낸때",
           -- 회원번호·닉네임을 그대로 주지 않습니다 — 번호 앞 8자만 (admin_공고신고목록 과 같은 꼴)
           left(q.profile_id::text, 8) as "낸사람"
      from "문의" q
     where (not "p_처리전만" or q."상태" <> '끝')
     order by q."낸때" desc
     limit 50 offset greatest("p_page", 0) * 50;
end $$;

-- ── 관리자 처리 ───────────────────────────────────────────────
create or replace function "admin_문의처리"("p_id" bigint, "p_상태" text, "p_메모" text default null)
returns void
language plpgsql volatile security definer set search_path to 'public'
as $$
begin
  if not "권한있나"('신고') then
    raise exception '권한이 없습니다' using errcode = '42501';
  end if;
  if "p_상태" not in ('받음', '처리중', '끝') then
    raise exception '상태는 받음·처리중·끝 중 하나입니다' using errcode = '22023';
  end if;
  update "문의"
     set "상태" = "p_상태",
         "처리메모" = coalesce(nullif(btrim("p_메모"), ''), "처리메모"),
         "처리한사람" = auth.uid(),
         "처리한때" = case when "p_상태" = '끝' then now() else "처리한때" end
   where id = "p_id";
  if not found then
    raise exception '그 문의가 없습니다' using errcode = 'P0002';
  end if;
end $$;

-- ── 권한 ──────────────────────────────────────────────────────
-- 새로 만든 함수는 PUBLIC 에게 열려 있습니다. 반드시 걷습니다 (지침 6-2 · 6-3)
revoke execute on function "문의넣기"(text, text, text, text)       from public, anon;
revoke execute on function "admin_문의목록"(boolean, int)            from public, anon;
revoke execute on function "admin_문의처리"(bigint, text, text)      from public, anon;
grant  execute on function "문의넣기"(text, text, text, text)        to authenticated;
grant  execute on function "admin_문의목록"(boolean, int)            to authenticated;
grant  execute on function "admin_문의처리"(bigint, text, text)      to authenticated;

commit;
```

## ㉡ 1:1 대화 창구 막기 (세중님 11/02 · 오픈 때 닫음)

```
왜      화면은 이미 닫았습니다 (web/lib/talk.ts 의 대화열렸나 = false · /talk·/talk/[room] 이
        안내만 그립니다). 그런데 **주소를 아는 사람이 회원 열쇠로 창구를 직접 부를 수는** 있습니다.
어떻게   함수 본문을 고치지 않고 **권한만 걷습니다** — 가장 작은 변경이고 한 줄로 되돌립니다.
        걷으면 회원이 부를 때 `permission denied for function …` 가 납니다
        (화면이 창구를 아예 안 부르니 회원에게는 안 보입니다)
위험     **낮음** — 그러나 **되돌릴 때 글자까지 같아야** 합니다. 전 ACL 을 아래에 찍어 뒀습니다
영향     관리자 쪽 「신고된 대화 열람」(권한 `대화열람`)은 **다른 길**입니다 —
        admin 함수가 service_role 로 돌아 영향 없습니다. 확인하고 올립니다
```
